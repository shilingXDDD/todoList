### 2.2 数据库版本 + 迁移 ✅ 已完成

📁 `app/src/main/java/com/example/todo_list/data/AppDatabase.kt`

```kotlin
@Database(entities = [Task::class], version = 2, exportSchema = false)
abstract class AppDatabase : RoomDatabase() {

    abstract fun taskDao(): TaskDao

    companion object {
        @Volatile
        private var INSTANCE: AppDatabase? = null

        val MIGRATION_1_2 = object : Migration(1, 2) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("ALTER TABLE tasks ADD COLUMN category TEXT NOT NULL DEFAULT '默认'")
                db.execSQL("ALTER TABLE tasks ADD COLUMN priority INTEGER NOT NULL DEFAULT 0")
                db.execSQL("ALTER TABLE tasks ADD COLUMN dueDate INTEGER")
                db.execSQL("ALTER TABLE tasks ADD COLUMN reminderTime INTEGER")
                db.execSQL("ALTER TABLE tasks ADD COLUMN note TEXT NOT NULL DEFAULT ''")
                db.execSQL("ALTER TABLE tasks ADD COLUMN link TEXT NOT NULL DEFAULT ''")
                db.execSQL("ALTER TABLE tasks ADD COLUMN createdAt INTEGER NOT NULL DEFAULT 0")
            }
        }

        fun getInstance(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "todo_database"
                )
                    .addMigrations(MIGRATION_1_2)
                    .build()
                    .also { INSTANCE = it }
            }
        }
    }
}
```

⚠️ 开发阶段不想写迁移可以用 `fallbackToDestructiveMigration()`，它会**清空数据重建表**。仅限开发期，答辩前必须去掉，否则"数据持久化"那 3 分会被质疑。

#### ② 新建 `StatisticsActivity.kt`

📁 `app/src/main/java/com/example/todo_list/ui/statistics/StatisticsActivity.kt`

```kotlin
package com.example.todo_list.ui.statistics

import android.os.Bundle
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.lifecycleScope
import androidx.lifecycle.repeatOnLifecycle
import com.example.todo_list.R
import com.example.todo_list.data.TaskRepository
import com.example.todo_list.databinding.ActivityStatisticsBinding
import kotlinx.coroutines.launch

class StatisticsActivity : AppCompatActivity() {

    private lateinit var binding: ActivityStatisticsBinding

    // 缓存最新值：两路 Flow 谁先返回，都能算出正确的未完成数与完成率
    private var total = 0
    private var completed = 0

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityStatisticsBinding.inflate(layoutInflater)
        setContentView(binding.root)

        supportActionBar?.setDisplayHomeAsUpEnabled(true)
        title = getString(R.string.title_statistics)

        observeStatistics()
    }

    private fun observeStatistics() {
        lifecycleScope.launch {
            // repeatOnLifecycle：页面退到后台自动停止收集，避免无谓刷新
            repeatOnLifecycle(Lifecycle.State.STARTED) {

                launch {
                    TaskRepository.countAll().collect {
                        total = it
                        binding.txvTotal.text = it.toString()
                        refreshDerived()
                    }
                }

                launch {
                    TaskRepository.countCompleted().collect {
                        completed = it
                        binding.txvCompleted.text = it.toString()
                        refreshDerived()
                    }
                }
            }
        }
    }

    /**
     * 未完成数与完成率都由 total / completed 推导。
     *
     * ⚠️ 必须先给字段赋值、再统一计算（本写法在两个 collect 里各自赋值后调用）。
     * 如果像早期版本那样在 A 的回调里传 B 的旧值，会出现短暂的负数：
     * countCompleted 先返回时，_total 还是 0，会算出 (0 - 2) = -2。
     */
    private fun refreshDerived() {
        binding.txvPending.text = (total - completed).toString()

        // ⚠️ 空列表时 total 为 0，直接除会得到 NaN，必须兜底
        val rate = if (total == 0) 0 else completed * 100 / total
        binding.txvRate.text = getString(R.string.stats_rate_value, rate)
    }

    override fun onSupportNavigateUp(): Boolean {
        finish()
        return true
    }
}
```

⚠️ **不要写成「在 A 的回调里传 B 的旧缓存值」**。早期版本 `updateRate(total, _completed)` 在
`countCompleted` 先返回时会算出 `0 - 2 = -2`，界面闪一下负数。现在的写法是先赋值再统一计算，不存在这个问题。

配套加一条文案：

```xml
<string name="stats_rate_value">%1$d%%</string>
```

⚠️ `%` 在 `strings.xml` 里要写成 `%%` 转义，否则编译报错。

**要点说明**：

| 写法 | 为什么 |
|---|---|
| `repeatOnLifecycle(STARTED)` | 页面不可见时停止收集，避免后台无谓刷新 |
| 两个 `launch` 并行收集 | 总数和完成数是两个独立 Flow，分开更清晰 |
| `_total` / `_completed` 缓存 | 两路 Flow 到达顺序不定，缓存后任意一路都能算出正确的"未完成" |
| `if (total == 0) 0` | 空列表时 `0/0` 会得到 `NaN`，必须兜底 |

```kotlin
        override fun onQueryTextSubmit(query: String?): Boolean = false

        override fun onQueryTextChange(newText: String?): Boolean {
            keyword = newText.orEmpty()
            applyFilter(latestTasks)      // 复用下面的过滤方法
            return true
        }
    })
}
```

把原来的 `observeTasks()` 改成这样：

```kotlin
private var latestTasks: List<Task> = emptyList()

private fun observeTasks() {
    viewLifecycleOwner.lifecycleScope.launch {
        repeatOnLifecycle(Lifecycle.State.STARTED) {
            TaskRepository.getTasks().collect { tasks ->
                latestTasks = tasks
                applyFilter(tasks)
            }
        }
    }
}

private fun applyFilter(allTasks: List<Task>) {
    val filtered = if (keyword.isBlank()) allTasks
                   else allTasks.filter { it.title.contains(keyword, true) }

    taskAdapter.submitList(filtered)

    // ⚠️ 评分表特别要求：无结果要有明确提示
    binding.emptyTextView.visibility = if (filtered.isEmpty()) View.VISIBLE else View.GONE
    binding.emptyTextView.text =
        if (keyword.isBlank()) getString(R.string.empty_task_list)
        else getString(R.string.empty_search_result)
}
```

**用「内存过滤」而不是查数据库**：逻辑简单、不用切换数据源，课设级别的数据量完全够用。

```kotlin
private enum class DateFilter { ALL, TODAY, OVERDUE }

private var currentFilter = DateFilter.ALL

private fun applyFilter(allTasks: List<Task>) {
    val now = System.currentTimeMillis()
    var filtered = allTasks

    // 先按关键词
    if (keyword.isNotBlank()) {
        filtered = filtered.filter { it.title.contains(keyword, true) }
    }

    // 再按日期
    filtered = when (currentFilter) {
        DateFilter.TODAY -> filtered.filter {
            it.dueDate != null && it.dueDate in todayStart()..todayEnd()
        }
        DateFilter.OVERDUE -> filtered.filter {
            !it.isCompleted && it.dueDate != null && it.dueDate < now
        }
        else -> filtered
    }

    taskAdapter.submitList(filtered)
    binding.emptyTextView.visibility = if (filtered.isEmpty()) View.VISIBLE else View.GONE
    binding.emptyTextView.text =
        if (keyword.isBlank() && currentFilter == DateFilter.ALL)
            getString(R.string.empty_task_list)
        else getString(R.string.empty_search_result)
}
```

⚠️ `it.dueDate` 是 `Long?`，**不能**直接参与 `in start..end` 比较，必须先判空，否则编译不过。

**④ "今天"的起止时间**：

```kotlin
private fun todayStart(): Long = Calendar.getInstance().apply {
    set(Calendar.HOUR_OF_DAY, 0); set(Calendar.MINUTE, 0)
    set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0)
}.timeInMillis

private fun todayEnd(): Long = Calendar.getInstance().apply {
    set(Calendar.HOUR_OF_DAY, 23); set(Calendar.MINUTE, 59)
    set(Calendar.SECOND, 59); set(Calendar.MILLISECOND, 999)
}.timeInMillis
```

⚠️ **必须用 `Calendar` 按当天 0 点/23:59:59 算**，不要拿 `24*60*60*1000` 加减 —— 跨天边界会错。

## 十、AlarmManager 提醒与取消（4 分，工作量最大）

⚠️ **建议最后做**，依赖多、坑多。

### 10.1 注册提醒

```kotlin
private fun setReminder(taskId: Long, timeMillis: Long, title: String) {
    val alarmManager = getSystemService(AlarmManager::class.java)

    val intent = Intent(this, TaskReceiver::class.java).apply {
        action = TaskReceiver.ACTION_TASK_REMIND
        putExtra(TaskReceiver.EXTRA_TASK_ID, taskId)
        putExtra(TaskReceiver.EXTRA_TASK_TITLE, title)
    }
    val pi = PendingIntent.getBroadcast(
        this, taskId.toInt(), intent,
        PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
    )

    alarmManager.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, timeMillis, pi)
}
```

⚠️ 用 **`getBroadcast`**（不是 `getActivity`），因为目标是 Receiver。

⚠️ Android 12+ 用 `setExact` 需要 `SCHEDULE_EXACT_ALARM` 权限，且**默认关闭**，要跳设置页让用户开。课设阶段建议用 `setWindow()` 或 `set()` 规避。

### 10.2 删除时取消提醒（2 分，必拿）

⚠️ **这 2 分最容易被忽略，也最容易在答辩演示时翻车**：

只删数据库不取消 Alarm → 到点弹出通知 → 用户点进去 → 任务已不存在 → **崩溃**。

```kotlin
private fun cancelReminder(taskId: Long) {
    val alarmManager = getSystemService(AlarmManager::class.java)
    val intent = Intent(this, TaskReceiver::class.java).apply {
        action = TaskReceiver.ACTION_TASK_REMIND
    }
    val pi = PendingIntent.getBroadcast(
        this, taskId.toInt(), intent,
        PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
    )
    alarmManager.cancel(pi)     // ← 关键
    pi.cancel()
}
```

**取消用的 Intent 必须和注册时"匹配"**：相同的 action、相同的 requestCode。
Kotlin 里 `Intent` 的 extra 不参与匹配，所以只放 action 也能取消掉。

在删除确认的 `PositiveButton` 里调用：

```kotlin
lifecycleScope.launch {
    TaskRepository.deleteTask(current)
    cancelReminder(current.id)      // ⚠️ 别漏
    finish()
}
```

| 现象 | 原因 | 解决 |
|---|---|---|
| 加字段后 App 崩溃 | 没 `version + 1` 或没写迁移 | 见第二节 |
| **编辑保存后分类/优先级全没了** | `Task(id, ...)` 重建对象丢了字段 | 用 `existing.copy(...)`，见 2.6 节 |
| 统计数永远是 0 | `WHERE isCompleted = true` | SQLite 要用 `= 1` |
| 完成率显示 `NaN` | 除数为 0 | 判断总数是否为 0 |
| 完成一直是 0% | 整数除法 `completed/total` 得 0 | 先乘 100 再除：`completed * 100 / total` |
| 搜索无结果但没提示 | 忘了改空状态文案 | 加 `empty_search_result` |
| 同日期任务排序错乱 | 毫秒没清零 | `set(Calendar.MILLISECOND, 0)` |
| 筛选点了没反应 | 用了废弃的监听器 | 用 `setOnCheckedStateChangeListener` |
| 已完成的任务还标红 | 过期判断漏了 `!isCompleted` | 补上 |
| `strings.xml` 编译报错 | `%` 没转义 | 写成 `%%` |
| 删了任务还收到通知 | 没取消 Alarm | `alarmManager.cancel(pi)` |
| PendingIntent 崩溃 | 缺 `FLAG_IMMUTABLE` | 补上 |
| `Uri.parse` 崩溃 | 链接没有 http 前缀 | 判断并补全 |
| SearchView 配色不对 | 用了 `android.widget.SearchView` | 换成 androidx 版 |
| **Adapter 里 `getString()` 爆红** | Adapter 不是 Context | 用 `holder.itemView.context.getString(...)` |
| **`binding.xxx` 爆红 Unresolved reference** | 布局里没这个 id，或 ViewBinding 没重新生成 | 检查 id 拼写；`Build → Rebuild Project` |
| `strings.xml` 编译报 not well-formed | 用了 `//` 注释 | XML 必须用 `<!-- -->` |
| Spinner 回显选中不了 | adapter 还没设就调 `setSelection()` | 先 `adapter = ...` 再 `setSelection()` |
| 日期差一个月 | `DatePickerDialog` 的 month 是 0~11 | 直接传给 `Calendar.set()`，不要 ±1 |

### A.3 本次踩到的三个坑（汇总）

| # | 坑 | 表现 | 修法 |
|---|---|---|---|
| 1 | `strings.xml` 用了 `//` 注释 | 整个项目编译失败 | 改成 `<!-- -->` |
