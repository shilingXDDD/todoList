## 三、第 2~3 天：Kotlin 代码

### 3.1 `data/Task.kt` ✏️

```kotlin
package com.example.todo_list.data

data class Task(
    val id: Long = 0,
    val title: String,
    val description: String = "",
    val isCompleted: Boolean = false
)
```

**为什么用 `data class`**：自动生成 `equals()`/`hashCode()`/`copy()`。后面 `DiffUtil` 要靠 `equals` 判断内容有没有变，`copy()` 用来"改一个字段、其余不变"。

**关于 `val` 而不是 `var`**：改状态时用 `task.copy(isCompleted = true)` 生成新对象，而不是改原对象。这是不可变（immutable）写法，能避免很多难查的 bug——比如列表里的数据被意外改掉但界面没刷新。

> 第二步加 Room 时，只在类名上加 `@Entity(tableName = "tasks")`、在 `id` 上加 `@PrimaryKey(autoGenerate = true)`，**这里的结构一个字都不用改**。

### 3.2 `data/TaskRepository.kt` ✏️

```kotlin
package com.example.todo_list.data

object TaskRepository {

    private val tasks = mutableListOf(
        Task(1, "完成 Android 课设", "包含 Room、Service、广播三个模块", false),
        Task(2, "复习 Fragment 生命周期", "onCreateView 与 onViewCreated 的区别", true),
        Task(3, "整理第一步笔记", "", false)
    )

    private var nextId = 4L

    /** 返回副本，避免外部直接改动内部列表 */
    fun getTasks(): List<Task> = tasks.toList()

    fun getTaskById(id: Long): Task? = tasks.find { it.id == id }

    fun addTask(task: Task): Long {
        val newTask = task.copy(id = nextId++)
        tasks.add(0, newTask)          // 加到最前面，用户能立刻看到
        return newTask.id
    }

    fun updateTask(task: Task) {
        val index = tasks.indexOfFirst { it.id == task.id }
        if (index != -1) {
            tasks[index] = task
        }
    }
}
```

> ⚠️ **为什么 `getTasks()` 要 `.toList()`**
>
> `toList()` 会创建一个**新列表**。如果直接返回 `tasks`，外部拿到的是同一个引用，后面 `submitList()` 传同一个对象时 DiffUtil 会认为"没变化"从而**不刷新**。这是新手最常踩的坑之一，表现为"数据改了但界面不动"。

**为什么要有 Repository 这一层**：第二步换 Room 时，只改这个文件内部实现，Fragment 和 Adapter **一行都不用动**。现在多写一个文件，后面省一堆事。

### 3.3 `ui/list/TaskAdapter.kt` ✏️

```kotlin
package com.example.todo_list.ui.list

import android.graphics.Paint
import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.example.todo_list.data.Task
import com.example.todo_list.databinding.ItemTaskBinding

class TaskAdapter(
    private val onItemClick: (Task) -> Unit,
    private val onCheckedChange: (Task, Boolean) -> Unit
) : ListAdapter<Task, TaskAdapter.ViewHolder>(DiffCallback) {

    class ViewHolder(val binding: ItemTaskBinding) :
        RecyclerView.ViewHolder(binding.root)

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ViewHolder {
        val binding = ItemTaskBinding.inflate(
            LayoutInflater.from(parent.context), parent, false
        )
        return ViewHolder(binding)
    }

    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val task = getItem(position)

        holder.binding.titleTextView.text = task.title

        if (task.description.isBlank()) {
            holder.binding.descriptionTextView.visibility = android.view.View.GONE
        } else {
            holder.binding.descriptionTextView.visibility = android.view.View.VISIBLE
            holder.binding.descriptionTextView.text = task.description
        }

        // 完成后标题加中划线
        holder.binding.titleTextView.paintFlags =
            if (task.isCompleted) {
                holder.binding.titleTextView.paintFlags or Paint.STRIKE_THRU_TEXT_FLAG
            } else {
                holder.binding.titleTextView.paintFlags and Paint.STRIKE_THRU_TEXT_FLAG.inv()
            }

        // 整行点击 → 进入编辑页
        holder.binding.root.setOnClickListener {
            onItemClick(task)
        }

        // ⚠️ 关键：先清空旧监听器，再设值，最后设新监听器
        holder.binding.completedCheckBox.setOnCheckedChangeListener(null)
        holder.binding.completedCheckBox.isChecked = task.isCompleted
        holder.binding.completedCheckBox.setOnCheckedChangeListener { _, isChecked ->
            onCheckedChange(task, isChecked)
        }
    }

    object DiffCallback : DiffUtil.ItemCallback<Task>() {
        override fun areItemsTheSame(oldItem: Task, newItem: Task): Boolean {
            return oldItem.id == newItem.id          // 是不是同一条数据
        }

        override fun areContentsTheSame(oldItem: Task, newItem: Task): Boolean {
            return oldItem == newItem                // 内容有没有变（靠 data class 的 equals）
        }
    }
}
```

> ⚠️ **CheckBox 的三行为什么这么写**
>
> RecyclerView 会**复用** ViewHolder。滚动到第 10 行时，用的可能是第 1 行那个 ViewHolder。如果不先 `setOnCheckedChangeListener(null)`，执行 `isChecked = task.isCompleted` 时会触发**旧的监听器**，把错误的状态写回去。
>
> 正确顺序永远是：**清空监听器 → 设值 → 设新监听器**。
>
> 这个 bug 的表现是"勾选了第 1 条，滚动几下发现第 8 条也自动勾上了"。现在多写一行，省掉后面两小时的排查。

**为什么用 `ListAdapter` 而不是 `RecyclerView.Adapter`**：`ListAdapter` 内置 DiffUtil，`submitList()` 时自动算差异、只刷新变化的行，还带动画。第二步接 Room 的 `Flow` 时直接 `submitList()` 就行。

### 3.4 `ui/list/TaskListFragment.kt` ✏️

```kotlin
package com.example.todo_list.ui.list

import android.content.Intent
import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.fragment.app.Fragment
import com.example.todo_list.data.TaskRepository
import com.example.todo_list.databinding.FragmentTaskListBinding
import com.example.todo_list.ui.edit.TaskEditActivity

class TaskListFragment : Fragment() {

    private var _binding: FragmentTaskListBinding? = null
    private val binding get() = _binding!!

    private lateinit var taskAdapter: TaskAdapter

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentTaskListBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        setupRecyclerView()

        binding.addTaskFab.setOnClickListener {
            // 不传 ID → 新增模式
            startActivity(Intent(requireContext(), TaskEditActivity::class.java))
        }
    }

    private fun setupRecyclerView() {
        taskAdapter = TaskAdapter(
            onItemClick = { task ->
                val intent = Intent(requireContext(), TaskEditActivity::class.java)
                    .putExtra(TaskEditActivity.EXTRA_TASK_ID, task.id)
                startActivity(intent)
            },
            onCheckedChange = { task, isChecked ->
                TaskRepository.updateTask(task.copy(isCompleted = isChecked))
                loadTasks()
            }
        )
        binding.taskRecyclerView.adapter = taskAdapter
    }

    override fun onResume() {
        super.onResume()
        // 从编辑页返回时会走到这里 → 列表自动刷新
        loadTasks()
    }

    private fun loadTasks() {
        val tasks = TaskRepository.getTasks()
        taskAdapter.submitList(tasks)
        binding.emptyTextView.visibility =
            if (tasks.isEmpty()) View.VISIBLE else View.GONE
    }

    override fun onDestroyView() {
        super.onDestroyView()
        _binding = null      // ⚠️ Fragment 必须置空，否则内存泄漏
    }
}
```

> ⚠️ **为什么 `onResume()` 里加载数据**
>
> 从编辑页 `finish()` 回到列表页时，Fragment 会走 `onResume()`。在这里重新取一次数据，就能保证"保存后返回列表能看到最新内容"——这正对应评分表"返回列表的数据刷新"**3 分**。
>
> 第二步换成 Room + Flow 之后，这行 `loadTasks()` 就可以删掉了，数据会自动推过来。但第一步先用这个最朴素可靠的办法把分拿到。

> ⚠️ **为什么 Fragment 的 binding 要可空 + 置空**
>
> Fragment 的 **View 生命周期**和 **Fragment 自身生命周期**不一样长。Fragment 还活着但 View 已经被销毁时（比如被另一个 Fragment 替换掉），`_binding` 还指向旧 View，此时访问 `binding.xxx` 就崩。
>
> `onDestroyView()` 里置空是标准解法，这也是评分表"避免在 Fragment View 销毁后继续访问 View"那 **1 分** 的标准答案。Activity 不需要这么做，因为它的 View 和自己是同生共死的。

### 3.5 `ui/edit/TaskEditActivity.kt` ✏️

```kotlin
package com.example.todo_list.ui.edit

import android.os.Bundle
import android.view.MenuItem
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.example.todo_list.R
import com.example.todo_list.data.Task
import com.example.todo_list.data.TaskRepository
import com.example.todo_list.databinding.ActivityTaskEditBinding

class TaskEditActivity : AppCompatActivity() {

    companion object {
        const val EXTRA_TASK_ID = "task_id"
        const val NO_TASK_ID = -1L
    }

    private lateinit var binding: ActivityTaskEditBinding
    private var taskId = NO_TASK_ID

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityTaskEditBinding.inflate(layoutInflater)
        setContentView(binding.root)

        setSupportActionBar(binding.toolbar)
        supportActionBar?.setDisplayHomeAsUpEnabled(true)   // 显示返回箭头

        // 取出列表页传来的 ID；取不到就是 -1，代表新增
        taskId = intent.getLongExtra(EXTRA_TASK_ID, NO_TASK_ID)

        if (taskId != NO_TASK_ID) {
            setupEditMode()
        } else {
            setupCreateMode()
        }

        binding.saveButton.setOnClickListener {
            saveTask()
        }
    }

    private fun setupEditMode() {
        val task = TaskRepository.getTaskById(taskId)
        if (task == null) {
            // ⚠️ 判空：ID 非法时不能崩
            Toast.makeText(this, R.string.msg_task_not_found, Toast.LENGTH_SHORT).show()
            finish()
            return
        }
        supportActionBar?.title = getString(R.string.title_edit_task)
        binding.titleEditText.setText(task.title)
        binding.descriptionEditText.setText(task.description)
        binding.completedCheckBox.isChecked = task.isCompleted
    }

    private fun setupCreateMode() {
        supportActionBar?.title = getString(R.string.title_add_task)
    }

    private fun saveTask() {
        // trim() 很重要：只输入空格也算空
        val title = binding.titleEditText.text?.toString()?.trim().orEmpty()
        if (title.isEmpty()) {
            binding.titleInputLayout.error = getString(R.string.msg_title_required)
            return                       // ⚠️ 直接 return，不保存不返回
        }
        binding.titleInputLayout.error = null   // 通过了就清掉错误提示

        val description = binding.descriptionEditText.text?.toString()?.trim().orEmpty()
        val isCompleted = binding.completedCheckBox.isChecked

        if (taskId != NO_TASK_ID) {
            TaskRepository.updateTask(
                Task(taskId, title, description, isCompleted)
            )
        } else {
            TaskRepository.addTask(
                Task(title = title, description = description, isCompleted = isCompleted)
            )
        }

        Toast.makeText(this, R.string.msg_save_success, Toast.LENGTH_SHORT).show()
        finish()      // 关闭当前页，回到列表页（会触发列表页的 onResume）
    }

    override fun onOptionsItemSelected(item: MenuItem): Boolean {
        if (item.itemId == android.R.id.home) {
            finish()
            return true
        }
        return super.onOptionsItemSelected(item)
    }
}
```

**这段代码的关键逻辑**：

| 逻辑 | 怎么实现 |
|---|---|
| 新增还是编辑？ | 看 `taskId` 是不是 `-1` |
| 编辑时回填数据 | `setText(task.title)` 等 |
| 标题空校验 | `trim()` 后判空，空则 `setError` + `return` |
| 保存后回列表 | `finish()` |
| 左上角返回箭头 | `setDisplayHomeAsUpEnabled(true)` + `onOptionsItemSelected` 处理 `android.R.id.home` |

> ⚠️ **`orEmpty()` 的作用**：`binding.titleEditText.text` 返回的是 `Editable?`（可能为空），`?.toString()?.trim()` 之后还是可能为 `null`。`orEmpty()` 保证最终一定是个 `String`，后面判空不会漏。

```kotlin
package com.example.todo_list

import android.os.Bundle
import androidx.appcompat.app.AppCompatActivity
import com.example.todo_list.databinding.ActivityMainBinding
import com.example.todo_list.ui.list.TaskListFragment

class MainActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityMainBinding.inflate(layoutInflater)
        setContentView(binding.root)

        // ⚠️ 只在第一次创建时添加 Fragment
        if (savedInstanceState == null) {
            supportFragmentManager.beginTransaction()
                .replace(R.id.fragmentContainer, TaskListFragment())
                .commit()
        }
    }
}
```

> ⚠️ **为什么要有 `if (savedInstanceState == null)`**
>
> 转屏时 Activity 会重建，但 FragmentManager 会**自动恢复**已经存在的 Fragment。如果不加这个判断，会再 `replace` 一次，导致屏幕上叠着两个 Fragment，列表出现两份数据。

**注意**：`MainActivity` 里**没有**任何数据库操作、没有业务逻辑，只负责装 Fragment。这就是评分表"Activity 职责清晰"那 3 分的拿法。

## 四、完整数据流转（逻辑串讲）

把上面的代码串起来看一次，理解数据怎么流动的：

```
【1】App 启动
     MainActivity.onCreate()
     → 把 TaskListFragment 放进 fragmentContainer

【2】列表页显示
     TaskListFragment.onViewCreated()  → 创建 Adapter，绑到 RecyclerView
     TaskListFragment.onResume()       → loadTasks()
                                        → TaskRepository.getTasks()  拿到 3 条假数据
                                        → adapter.submitList(tasks)  RecyclerView 渲染 3 行

【3】点右下角 FAB
     startActivity(TaskEditActivity)   ← 没有 putExtra，所以 taskId = -1
     → TaskEditActivity.setupCreateMode()  标题显示"新建任务"，输入框空的

【4】用户填内容，点保存
     saveTask()
     → 取标题，trim 后判空
     → 空 → titleInputLayout.error 显示红字，return（不保存不返回）
     → 非空 → TaskRepository.addTask(...)
     → Toast "保存成功"
     → finish()   ← 关闭编辑页

【5】回到列表页
     TaskListFragment.onResume() 被触发
     → loadTasks() 重新取数据 → 这次是 4 条
     → submitList() → 新任务出现在列表最上面   ✅ 这就是"返回刷新"3 分

【6】点列表某一行
     TaskAdapter 的 onItemClick(task) 被回调
     → Intent 里 putExtra("task_id", task.id)   ← 比如 id = 2
     → TaskEditActivity 里 getLongExtra 拿到 2
     → setupEditMode() → getTaskById(2) 查出任务 → setText 回填
     → 用户改完点保存 → updateTask() → finish() → 回到列表自动刷新

【7】勾选某一行的 CheckBox
     TaskAdapter 的 onCheckedChange(task, true) 被回调
     → TaskRepository.updateTask(task.copy(isCompleted = true))
     → loadTasks() 重新渲染 → 该行标题加中划线
```

**一句话总结**：`TaskRepository` 是唯一的数据源，页面不持有数据，每次要显示就去取一次。这样第二步换成 Room 时，只要让 `getTasks()` 从数据库取，所有页面自动跟着变。

## 五、几个"为什么这么写"

**为什么传 `task_id` 而不传整个 Task 对象？**
传对象要让 `Task` 实现 `Serializable`/`Parcelable`，多一层出错可能；而且传过去的是**快照**，改完还得想办法传回来。传 ID 是标准做法，接收方自己查，永远是最新数据。

**为什么 `Task` 用 `val` 不用 `var`？**
改状态用 `task.copy(isCompleted = true)` 生成新对象。DiffUtil 靠 `equals` 比较新旧对象，如果直接改原对象的字段，新旧是同一个引用，DiffUtil 判断"没变化"，界面就不刷新。

**为什么 `getTasks()` 返回的是 `List` 副本？**
同上。返回同一个 mutable list，`submitList` 传同一个对象，DiffUtil 认为没变化，不刷新。

**为什么用 `ListAdapter` 不用 `RecyclerView.Adapter`？**
自带 DiffUtil，`submitList` 自动算差异、带增删动画。第二步接 Flow 时零成本。

**为什么 `onResume` 加载而不是 `onCreateView`？**
`onCreateView` 只在创建时走一次；`onResume` 每次回到页面都走。从编辑页返回时需要重新加载，必须放 `onResume`。

## 六、常见错误与排查

| 报错 / 现象 | 原因 | 解决 |
|---|---|---|
| `Unresolved reference: ActivityTaskEditBinding` | 布局文件名拼错，或没 Sync | 检查文件名是否 `activity_task_edit.xml`，然后 Build → Clean / Sync |
| `ActivityTaskEditBinding` 找不到但 Sync 过了 | `buildFeatures` 没开 | 检查 2.1 |
| 编译报 `android:exported must be explicitly specified` | Manifest 漏了 | 见 3.7 |
| 运行时 `ActivityNotFoundException` | Manifest 没注册，或类名路径写错 | 见 3.7 |
| 保存后返回，列表没变化 | `getTasks()` 返回了同一个 list 引用 | 确认有 `.toList()` |
| 勾选第 1 条，滚动后第 8 条也勾上了 | CheckBox 没清空旧监听器 | 见 3.3 的三行 |
| 转屏后列表变两份 | 没加 `savedInstanceState == null` 判断 | 见 3.6 |

| 错误提示不显示 | 用了 `titleEditText.error` | 改用 `titleInputLayout.error` |
| `lateinit property binding has not been initialized` | 在赋值前访问了 binding | 确保 `onCreate` 第一行就 `inflate` |
